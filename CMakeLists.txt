cmake_minimum_required(VERSION 3.20)
project(fileindex VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DUCKDB_SDK "" CACHE PATH "DuckDB 1.4.1 SDK (library and headers)")
set(DUCKDB_SOURCE "" CACHE PATH "DuckDB v1.4.1 source tree")
set(DUCKDB_STATIC_LIBRARY "" CACHE FILEPATH "Optional prebuilt DuckDB 1.4.1 Windows static library")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(fileindex SHARED src/fileindex.cpp)
target_include_directories(fileindex SYSTEM PRIVATE "${DUCKDB_SOURCE}/src/include")
target_compile_definitions(fileindex PRIVATE DUCKDB_BUILD_LOADABLE_EXTENSION DUCKDB_STATIC_BUILD)
set_target_properties(fileindex PROPERTIES PREFIX "" SUFFIX ".duckdb_extension")
set_property(TARGET fileindex APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/metadata.py")
if(MSVC)
  if(NOT DUCKDB_STATIC_LIBRARY)
    add_library(fileindex_duckdb STATIC "${DUCKDB_SDK}/amalgamation/duckdb.cpp")
    target_include_directories(fileindex_duckdb PRIVATE "${DUCKDB_SDK}/amalgamation")
    target_compile_definitions(fileindex_duckdb PRIVATE DUCKDB_STATIC_BUILD DUCKDB_BUILD_LIBRARY NOMINMAX WIN32_LEAN_AND_MEAN NOGDI)
    target_compile_options(fileindex_duckdb PRIVATE /bigobj /EHsc /utf-8)
    set(DUCKDB_STATIC_LIBRARY fileindex_duckdb)
  endif()
  target_link_libraries(fileindex PRIVATE ${DUCKDB_STATIC_LIBRARY} ws2_32 rstrtmgr bcrypt)
  target_compile_options(fileindex PRIVATE /W4 /WX /EHsc /utf-8 /external:W0)
  set(FILEINDEX_PLATFORM windows_amd64)
else()
  file(GLOB DUCKDB_ARCHIVES "${DUCKDB_SDK}/linux-static/*.a")
  if(NOT DUCKDB_ARCHIVES)
    message(FATAL_ERROR "Official DuckDB static archives are required")
  endif()
  target_link_libraries(fileindex PRIVATE -Wl,--start-group ${DUCKDB_ARCHIVES} -Wl,--end-group dl pthread)
  target_link_options(fileindex PRIVATE -Wl,--exclude-libs,ALL -Wl,--gc-sections)
  set_target_properties(fileindex PROPERTIES CXX_VISIBILITY_PRESET hidden)
  target_compile_options(fileindex PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -fno-rtti)
  set(FILEINDEX_PLATFORM linux_amd64)
endif()
add_custom_command(TARGET fileindex POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/metadata.py"
          "$<TARGET_FILE:fileindex>" "${FILEINDEX_PLATFORM}")
